        let currentFilter = "all";

        function renderProductCard(p) {
            const badgeClass = p.badge === 'hot' ? 'badge-hot' : (p.badge === 'new' ? 'badge-new' : (p.badge === 'promo' ? 'badge-promo' : 'badge-sale'));
            const badgeText = p.badge === 'hot' ? 'TOP' : (p.badge === 'new' ? 'NEW' : (p.badge === 'promo' ? 'PROMO' : 'SALE'));
            return `
                <div class="product-card" data-id="${p.id}">
                    <div class="product-badge ${badgeClass}">${badgeText}</div>
                    <img src="${p.image}" alt="${p.name}" class="product-img" loading="lazy">
                    <div class="product-info">
                        <h3 class="product-title">${p.name.substring(0, 30)}</h3>
                        <div class="product-category">${p.category.toUpperCase()}</div>
                        <div class="product-price">${p.price.toLocaleString()} FCFA <del>${p.oldPrice.toLocaleString()} FCFA</del></div>
                        <div class="product-actions">
                            <button class="btn-detail" onclick='openDetailModal(${JSON.stringify(p).replace(/'/g, "&#39;")})'><i class="fas fa-eye"></i> Details</button>
                            <button class="btn-buy" onclick='openOrderModal("${p.id}", "${p.name.replace(/'/g, "\\'")}", ${p.price})'><i class="fas fa-cart-plus"></i> Acheter</button>
                        </div>
                    </div>
                </div>
            `;
        }

        function filterProducts(cat) {
            if(cat === 'all') return allProducts;
            return allProducts.filter(p => p.category === cat);
        }

        function renderAll() {
            const filtered = filterProducts(currentFilter);
            document.getElementById('topProductsGrid').innerHTML = filtered.slice(0,4).map(renderProductCard).join('');
            document.getElementById('horizontalProductsGrid').innerHTML = filtered.slice(4,9).map(renderProductCard).join('');
            document.getElementById('discoverGrid').innerHTML = filtered.slice(9,14).map(renderProductCard).join('');
            const promoMix = filtered.filter(p => p.badge === 'hot' || p.badge === 'promo').slice(0,4);
            while(promoMix.length < 4) promoMix.push(filtered[0]);
            document.getElementById('mixProductsGrid').innerHTML = promoMix.map(renderProductCard).join('');
        }

        // MODAL DETAILS
        function openDetailModal(p) {
            const specs = p.specs.map(s => `<li><i class="fas fa-check-circle" style="color:var(--primary);"></i> ${s}</li>`).join('');
            document.getElementById('detailTitle').innerHTML = p.name;
            document.getElementById('detailBody').innerHTML = `

                    <img src="${p.image}" style="width:180px; border-radius:16px;">
                    <div style="flex:1;">
                        <h3>${p.name}</h3>
                        <div style="font-size:1.3rem; color:var(--primary); font-weight:bold;">${p.price.toLocaleString()} FCFA <del>${p.oldPrice.toLocaleString()} FCFA</del></div>
                        <ul style="list-style:none; margin:1rem 0;">${specs}</ul>
                        <button class="btn-buy" onclick='closeDetailModal(); openOrderModal("${p.id}", "${p.name.replace(/'/g, "\\'")}", ${p.price})' style="width:100%;">Commander</button>
                    </div>
                </div>
            `;
            document.getElementById('detailModal').style.display = 'flex';
        }
